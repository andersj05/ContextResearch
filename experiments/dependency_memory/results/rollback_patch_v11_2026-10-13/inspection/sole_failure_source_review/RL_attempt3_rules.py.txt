from itertools import product, combinations
import heapq


def _ledger(priority, tax_base, refunds):
    def solve(data):
        chosen = {}
        cutoff = data.get("cutoff", 10**9)
        for pos, event in enumerate(data["events"]):
            sequence = event.get("sequence", 0)
            if sequence > cutoff:
                continue
            event_id = event.get("id", "anonymous:" + str(pos))
            if priority == "sequence_first":
                rank = (sequence, event.get("revision", 0), pos)
            else:
                rank = (event.get("revision", 0), pos)
            if event_id not in chosen or rank > chosen[event_id][0]:
                chosen[event_id] = (rank, event)

        balances = dict(data.get("opening", {}))
        half_up = data["rounding"] == "half_up"

        def rounded(numerator):
            if half_up:
                return (numerator + 5000) // 10000
            return numerator // 10000

        for _, event in chosen.values():
            if event.get("void", False):
                continue
            raw = event["price"] * event.get("quantity", 1)
            discount = min(10000, max(0, event.get("discount_bp", 0)))
            net = rounded(raw * (10000 - discount))
            base = raw if tax_base == "original" else net
            total = net + rounded(base * event.get("tax_bp", 0))
            if refunds and event.get("kind") == "refund":
                total = -total
            key = (event.get("account", "default") + "|" +
                   event.get("currency", "USD"))
            balances[key] = balances.get(key, 0) + total
        return {"balances": balances}
    return solve


def _dependencies():
    def solve(data):
        packages = data["packages"]
        names = sorted(packages, reverse=True)
        roots = data["roots"]
        features = set(data.get("features", []))
        blocked = data.get("blocked", {})
        pins = data.get("pins", {})
        choices = []
        for name in names:
            rows = [row for row in packages[name]
                    if row["version"] not in blocked.get(name, [])
                    and (name not in pins or row["version"] == pins[name])]
            choices.append([None] + rows)

        best_vector = None
        best_versions = None
        for assignment in product(*choices):
            installed = {name: row for name, row in zip(names, assignment)
                         if row is not None}
            if any(name not in installed or
                   not low <= installed[name]["version"] <= high
                   for name, (low, high) in roots.items()):
                continue

            edges = {}
            valid = True
            for name, row in installed.items():
                dependencies = []
                for dep in row.get("deps", []):
                    feature = dep.get("feature", "")
                    if feature and feature not in features:
                        continue
                    other = dep["name"]
                    if (other not in installed or
                            not dep["min"] <= installed[other]["version"] <= dep["max"]):
                        valid = False
                        break
                    dependencies.append(other)
                edges[name] = dependencies
                if not valid:
                    break
            if not valid:
                continue

            closure = set()
            stack = list(roots)
            while stack:
                name = stack.pop()
                if name in closure:
                    continue
                if name not in installed:
                    valid = False
                    break
                closure.add(name)
                stack.extend(edges[name])
            if not valid or closure != set(installed):
                continue

            finished, active = set(), set()

            def visit(name):
                if name in active:
                    return False
                if name in finished:
                    return True
                active.add(name)
                for other in edges[name]:
                    if not visit(other):
                        return False
                active.remove(name)
                finished.add(name)
                return True

            if not all(visit(name) for name in installed):
                continue

            vector = tuple(installed[name]["version"] if name in installed else -1
                           for name in names)
            if best_vector is None or vector > best_vector:
                best_vector = vector
                best_versions = {name: row["version"]
                                 for name, row in installed.items()}

        if best_versions is None:
            return {"error": "unsatisfiable"}
        return {"versions": best_versions}
    return solve


def _intervals():
    def solve(data):
        jobs = [job for job in data["jobs"]
                if job["id"] not in data.get("blocked", [])]
        k = data["k"]
        mandatory = set(data.get("mandatory", []))
        quotas = data.get("quotas", {})
        best_value = None
        best_ids = None

        for subset in combinations(jobs, k):
            ids = tuple(sorted(job["id"] for job in subset))
            if not mandatory.issubset(ids):
                continue
            counts = {}
            for job in subset:
                counts[job["group"]] = counts.get(job["group"], 0) + 1
            if any(not low <= counts.get(group, 0) <= high
                   for group, (low, high) in quotas.items()):
                continue

            ends = []
            feasible = True
            for job in sorted(subset, key=lambda item: (item["start"], item["end"])):
                while ends and ends[0] <= job["start"]:
                    heapq.heappop(ends)
                heapq.heappush(ends, job["end"])
                if len(ends) > 2:
                    feasible = False
                    break
            if not feasible:
                continue

            value = sum(job["value"] for job in subset)
            if (best_value is None or value > best_value or
                    (value == best_value and ids > best_ids)):
                best_value, best_ids = value, ids

        if best_value is None:
            return {"error": "impossible"}
        return {"selected": list(best_ids), "value": best_value}
    return solve


def _routes():
    def solve(data):
        required = data.get("required", [])
        bit_for = {node: 1 << index for index, node in enumerate(required)}
        full = (1 << len(required)) - 1
        source = data["source"]
        start_mask = bit_for.get(source, 0)
        start_state = (source, start_mask, data["coupons"])
        distance = {start_state: data["start"]}
        queue = [(data["start"], source, start_mask, data["coupons"])]
        graph = [[] for _ in range(data["n"])]
        for edge in data["edges"]:
            graph[edge["u"]].append(edge)

        while queue:
            time, node, mask, remaining = heapq.heappop(queue)
            state = (node, mask, remaining)
            if distance.get(state) != time:
                continue
            if node == data["target"] and mask == full:
                return {"arrival": time}
            for edge in graph[node]:
                depart = max(time, edge["open"])
                if depart >= edge["close"]:
                    continue
                options = [(edge["duration"], remaining)]
                if remaining:
                    options.append((max(1, edge["duration"] - edge.get("discount", 0)),
                                    remaining - 1))
                for duration, next_remaining in options:
                    arrival = depart + duration
                    if arrival > data["deadline"]:
                        continue
                    next_node = edge["v"]
                    next_mask = mask | bit_for.get(next_node, 0)
                    next_state = (next_node, next_mask, next_remaining)
                    if arrival < distance.get(next_state, float("inf")):
                        distance[next_state] = arrival
                        heapq.heappush(queue, (arrival, next_node, next_mask,
                                               next_remaining))
        return {"arrival": None}
    return solve
