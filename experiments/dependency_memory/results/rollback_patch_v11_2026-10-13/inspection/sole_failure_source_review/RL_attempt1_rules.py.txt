from itertools import product, combinations
import heapq


def _ledger(revision_priority, tax_base, refunds):
    def solve(data):
        cutoff = data.get("cutoff", 10**9)
        chosen = {}
        for position, event in enumerate(data["events"]):
            sequence = event.get("sequence", 0)
            if sequence > cutoff:
                continue
            event_id = event.get("id", "anonymous:" + str(position))
            if revision_priority == "sequence_first":
                rank = (sequence, event.get("revision", 0), position)
            else:
                rank = (event.get("revision", 0), position)
            if event_id not in chosen or rank > chosen[event_id][0]:
                chosen[event_id] = (rank, event)

        balances = dict(data.get("opening", {}))
        mode = data["rounding"]

        def rounded(n, d=10000):
            if mode == "half_up":
                return (n + d // 2) // d
            return n // d

        for _, event in chosen.values():
            if event.get("void", False):
                continue
            raw = event["price"] * event.get("quantity", 1)
            discount = min(10000, max(0, event.get("discount_bp", 0)))
            net = rounded(raw * (10000 - discount))
            base = raw if tax_base == "original" else net
            total = net + rounded(base * event.get("tax_bp", 0))
            if refunds and event.get("kind") == "refund":
                total = -total
            key = event.get("account", "default") + "|" + event.get("currency", "USD")
            balances[key] = balances.get(key, 0) + total
        return {"balances": balances}
    return solve


def _build_component_06():
    def solve(data):
        packages = data["packages"]
        names = sorted(packages)
        features = set(data.get("features", []))
        roots = data["roots"]
        blocked = data.get("blocked", {})
        pins = data.get("pins", {})
        choices = []
        for name in names:
            rows = [r for r in packages[name]
                    if r["version"] not in blocked.get(name, [])
                    and (name not in pins or r["version"] == pins[name])]
            choices.append([None] + rows)

        best_vector = None
        best_install = None
        for assignment in product(*choices):
            installed = {name: row for name, row in zip(names, assignment)
                         if row is not None}
            if any(name not in installed or not low <= installed[name]["version"] <= high
                   for name, (low, high) in roots.items()):
                continue

            edges = {}
            valid = True
            for name, row in installed.items():
                active = []
                for dep in row.get("deps", []):
                    feature = dep.get("feature", "")
                    if feature and feature not in features:
                        continue
                    other = dep["name"]
                    if (other not in installed or
                            not dep["min"] <= installed[other]["version"] <= dep["max"]):
                        valid = False
                        break
                    active.append(other)
                edges[name] = active
                if not valid:
                    break
            if not valid:
                continue

            closure = set()
            stack = list(roots)
            while stack:
                name = stack.pop()
                if name in closure:
                    continue
                if name not in installed:
                    valid = False
                    break
                closure.add(name)
                stack.extend(edges[name])
            if not valid or closure != set(installed):
                continue

            finished, active = set(), set()
            def visit(name):
                if name in active:
                    return False
                if name in finished:
                    return True
                active.add(name)
                if any(not visit(other) for other in edges[name]):
                    return False
                active.remove(name)
                finished.add(name)
                return True
            if not all(visit(name) for name in installed):
                continue

            vector = tuple(installed[name]["version"] if name in installed else -1
                           for name in names)
            if best_vector is None or vector > best_vector:
                best_vector = vector
                best_install = {name: row["version"] for name, row in installed.items()}

        if best_install is None:
            return {"error": "unsatisfiable"}
        return {"versions": best_install}
    return solve


def _build_component_07():
    def solve(data):
        jobs = [j for j in data["jobs"] if j["id"] not in data.get("blocked", [])]
        k = data["k"]
        mandatory = set(data.get("mandatory", []))
        quotas = data.get("quotas", {})
        best_value = None
        best_ids = None

        for subset in combinations(jobs, k):
            ids = tuple(sorted(j["id"] for j in subset))
            if not mandatory.issubset(ids):
                continue
            counts = {}
            for job in subset:
                counts[job["group"]] = counts.get(job["group"], 0) + 1
            if any(not low <= counts.get(group, 0) <= high
                   for group, (low, high) in quotas.items()):
                continue

            active_ends = []
            valid = True
            for job in sorted(subset, key=lambda j: (j["start"], j["end"])):
                while active_ends and active_ends[0] <= job["start"]:
                    heapq.heappop(active_ends)
                heapq.heappush(active_ends, job["end"])
                if len(active_ends) > 2:
                    valid = False
                    break
            if not valid:
                continue

            value = sum(j["value"] for j in subset)
            if (best_value is None or value > best_value or
                    (value == best_value and ids > best_ids)):
                best_value, best_ids = value, ids

        if best_value is None:
            return {"error": "impossible"}
        return {"selected": list(best_ids), "value": best_value}
    return solve


def _build_component_08():
    def solve(data):
        required = data.get("required", [])
        bit_for = {node: 1 << i for i, node in enumerate(required)}
        full = (1 << len(required)) - 1
        start_node = data["source"]
        start_mask = bit_for.get(start_node, 0)
        start_state = (start_node, start_mask, data["coupons"])
        dist = {start_state: data["start"]}
        queue = [(data["start"], start_node, start_mask, data["coupons"])]
        graph = [[] for _ in range(data["n"])]
        for edge in data["edges"]:
            graph[edge["u"]].append(edge)

        while queue:
            time, node, mask, remaining = heapq.heappop(queue)
            state = (node, mask, remaining)
            if dist.get(state) != time:
                continue
            if node == data["target"] and mask == full:
                return {"arrival": time}
            for edge in graph[node]:
                depart = max(time, edge["open"])
                if depart >= edge["close"]:
                    continue
                durations = [(edge["duration"], remaining)]
                if remaining:
                    durations.append((max(1, edge["duration"] - edge.get("discount", 0)), remaining - 1))
                for duration, next_remaining in durations:
                    arrival = depart + duration
                    if arrival > data["deadline"]:
                        continue
                    next_node = edge["v"]
                    next_mask = mask | bit_for.get(next_node, 0)
                    next_state = (next_node, next_mask, next_remaining)
                    if arrival < dist.get(next_state, float("inf")):
                        dist[next_state] = arrival
                        heapq.heappush(queue, (arrival, next_node, next_mask, next_remaining))
        return {"arrival": None}
    return solve
