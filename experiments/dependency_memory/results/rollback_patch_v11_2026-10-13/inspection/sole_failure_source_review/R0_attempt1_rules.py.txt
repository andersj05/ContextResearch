def _ledger(revision_first, discounted_tax):
    def rounded(n, d, mode):
        if mode == "half_up":
            return (n + d // 2) // d
        return n // d

    def solve(data):
        cutoff = data.get("cutoff", 10**9)
        chosen = {}
        for pos, event in enumerate(data["events"]):
            seq = event.get("sequence", 0)
            if seq > cutoff:
                continue
            eid = event.get("id", "anonymous:" + str(pos))
            key = ((event.get("revision", 0), pos) if revision_first else
                   (seq, event.get("revision", 0), pos))
            if eid not in chosen or key > chosen[eid][0]:
                chosen[eid] = (key, event)

        balances = dict(data.get("opening", {}))
        for _, event in chosen.values():
            if event.get("void", False):
                continue
            raw = event["price"] * event.get("quantity", 1)
            discount = min(10000, max(0, event.get("discount_bp", 0)))
            net = rounded(raw * (10000 - discount), 10000, data["rounding"])
            base = net if discounted_tax else raw
            tax = rounded(base * event.get("tax_bp", 0), 10000, data["rounding"])
            total = net + tax
            if event.get("kind") == "refund":
                total = -total
            key = event.get("account", "default") + "|" + event.get("currency", "USD")
            balances[key] = balances.get(key, 0) + total
        return {"balances": balances}
    return solve


def _build_component_06():
    from itertools import product

    def solve(data):
        packages = data["packages"]
        names = sorted(packages, reverse=True)
        features = set(data.get("features", []))
        roots = data["roots"]
        blocked = data.get("blocked", {})
        pins = data.get("pins", {})
        domains = []
        for name in names:
            versions = [row["version"] for row in packages[name]
                        if row["version"] not in blocked.get(name, []) and
                        (name not in pins or row["version"] == pins[name])]
            domains.append([None] + versions)
        rows = {name: {row["version"]: row for row in packages[name]} for name in names}
        best_vector = None
        best = None

        for choices in product(*domains):
            assignment = dict(zip(names, choices))
            installed = {name for name, version in assignment.items() if version is not None}
            if any(name not in installed or not bounds[0] <= assignment[name] <= bounds[1]
                   for name, bounds in roots.items()):
                continue
            edges = {}
            closure = set(roots)
            pending = list(roots)
            valid = True
            while pending and valid:
                name = pending.pop()
                version = assignment.get(name)
                if version is None:
                    valid = False
                    break
                deps = rows[name][version].get("deps", [])
                active_deps = [d for d in deps
                               if not d.get("feature") or d["feature"] in features]
                edges[name] = []
                for dep in active_deps:
                    other = dep["name"]
                    edges[name].append(other)
                    other_version = assignment.get(other)
                    if other_version is None or not dep["min"] <= other_version <= dep["max"]:
                        valid = False
                        break
                    if other not in closure:
                        closure.add(other)
                        pending.append(other)
            if not valid or closure != installed:
                continue

            finished, active = set(), set()
            def visit(name):
                if name in active:
                    return False
                if name in finished:
                    return True
                active.add(name)
                if any(not visit(other) for other in edges.get(name, [])):
                    return False
                active.remove(name)
                finished.add(name)
                return True
            if not all(visit(name) for name in installed):
                continue
            vector = tuple(-1 if assignment[name] is None else assignment[name]
                           for name in names)
            if best_vector is None or vector > best_vector:
                best_vector, best = vector, {name: assignment[name] for name in names
                                             if assignment[name] is not None}
        return {"versions": best} if best is not None else {"error": "unsatisfiable"}
    return solve


def _build_component_07():
    from itertools import combinations

    def solve(data):
        jobs = [job for job in data["jobs"]
                if job["id"] not in data.get("blocked", [])]
        mandatory = set(data.get("mandatory", []))
        quotas = data.get("quotas", {})
        k = data["k"]
        best_value = None
        best_ids = None
        for subset in combinations(jobs, k):
            ids = [job["id"] for job in subset]
            if not mandatory.issubset(ids):
                continue
            counts = {}
            for job in subset:
                counts[job["group"]] = counts.get(job["group"], 0) + 1
            if any(not low <= counts.get(group, 0) <= high
                   for group, (low, high) in quotas.items()):
                continue
            ordered = sorted(subset, key=lambda job: (job["start"], job["end"], job["id"]))
            active = []
            valid = True
            for job in ordered:
                active = [end for end in active if end > job["start"]]
                active.append(job["end"])
                if len(active) > 2:
                    valid = False
                    break
            if not valid:
                continue
            value = sum(job["value"] for job in subset)
            selected = sorted(ids)
            if (best_value is None or value > best_value or
                    (value == best_value and selected > best_ids)):
                best_value, best_ids = value, selected
        if best_value is None:
            return {"error": "impossible"}
        return {"selected": best_ids, "value": best_value}
    return solve


def _build_component_08():
    import heapq

    def solve(data):
        required = data.get("required", [])
        bit = {node: 1 << i for i, node in enumerate(required)}
        full = (1 << len(required)) - 1
        graph = [[] for _ in range(data["n"])]
        for edge in data["edges"]:
            graph[edge["u"]].append(edge)
        source, start = data["source"], data["start"]
        initial_mask = bit.get(source, 0)
        initial = (source, initial_mask, data["coupons"])
        dist = {initial: start}
        queue = [(start, source, initial_mask, data["coupons"])]
        while queue:
            time, node, mask, left = heapq.heappop(queue)
            state = (node, mask, left)
            if dist.get(state) != time:
                continue
            if node == data["target"] and mask == full:
                return {"arrival": time}
            for edge in graph[node]:
                depart = max(time, edge["open"])
                if depart >= edge["close"]:
                    continue
                options = [(edge["duration"], left)]
                if left:
                    options.append((max(1, edge["duration"] - edge.get("discount", 0)), left - 1))
                for duration, remaining in options:
                    arrival = depart + duration
                    if arrival > data["deadline"]:
                        continue
                    nxt = edge["v"]
                    next_mask = mask | bit.get(nxt, 0)
                    next_state = (nxt, next_mask, remaining)
                    if arrival < dist.get(next_state, float("inf")):
                        dist[next_state] = arrival
                        heapq.heappush(queue, (arrival, nxt, next_mask, remaining))
        return {"arrival": None}
    return solve
